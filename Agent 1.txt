```python
# Automated tests for the Simple Calculator module
import unittest

# The calculator functions would typically be imported from the calculator module.
# For this test, copy the functions here for direct testing.

def add(x, y):
    return x + y

def subtract(x, y):
    return x - y

def multiply(x, y):
    return x * y

def divide(x, y):
    if y == 0:
        return "Error! Division by zero."
    return x / y

class TestCalculator(unittest.TestCase):
    def test_add(self):
        self.assertEqual(add(2, 3), 5)
        self.assertEqual(add(-1, 1), 0)
        self.assertEqual(add(0, 0), 0)

    def test_subtract(self):
        self.assertEqual(subtract(10, 5), 5)
        self.assertEqual(subtract(-1, -1), 0)
        self.assertEqual(subtract(0, 5), -5)

    def test_multiply(self):
        self.assertEqual(multiply(3, 4), 12)
        self.assertEqual(multiply(-2, 3), -6)
        self.assertEqual(multiply(0, 100), 0)

    def test_divide(self):
        self.assertEqual(divide(10, 2), 5)
        self.assertEqual(divide(-9, 3), -3)
        self.assertEqual(divide(0, 1), 0)
        self.assertEqual(divide(5, 0), "Error! Division by zero.")

    def test_divide_float(self):
        self.assertAlmostEqual(divide(7, 2), 3.5)

if __name__ == '__main__':
    unittest.main()
```
